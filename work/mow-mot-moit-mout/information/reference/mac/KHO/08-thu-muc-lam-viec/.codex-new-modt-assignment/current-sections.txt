  function renderConditionValueRows(fieldCode, query) {
    var needle = String(query || '').trim().toLowerCase();
    var field = CONDITION_FIELD_CATALOG.find(function (item) {
      return item.code === fieldCode;
    });
    var values = field
      ? [{
          code: 'ALL',
          name: 'Tất cả ' + field.name.toLowerCase(),
          path: 'Giá trị hệ thống · áp dụng cho mọi giá trị hợp lệ',
          isAll: true
        }].concat(CONDITION_VALUE_SAMPLE[fieldCode] || [])
      : [];
    values = values.filter(function (value) {
      return !needle
        || value.code.toLowerCase().indexOf(needle) !== -1
        || value.name.toLowerCase().indexOf(needle) !== -1
        || value.path.toLowerCase().indexOf(needle) !== -1;
    });
    if (!values.length) {
      return '<tr><td colspan="3" class="new-modt-condition-picker__empty">Không tìm thấy giá trị phù hợp</td></tr>';
    }
    return values.map(function (value) {
      return '<tr class="new-modt-condition-picker__row'
        + (value.isAll ? ' new-modt-condition-picker__row--all' : '')
        + '" data-value-code="' + esc(value.code) + '"'
        + ' data-value-name="' + esc(value.name) + '"><td'
        + (value.isAll ? ' class="new-modt-condition-picker__all-code"' : '')
        + '>' + esc(value.code) + '</td>'
        + '<td>' + esc(value.name) + '</td><td class="new-modt-condition-picker__muted">'
        + esc(value.path) + '</td></tr>';
    }).join('');
  }

  function mount(root, task) {
    if (!root || root.querySelector(':scope > .new-modt-task-config')) return;
    ensureStyle();
    var taskKey = task && (task.code || task.title) ? String(task.code || task.title) : 'T1';
    if (!openState[taskKey]) openState[taskKey] = Object.create(null);

    var host = document.createElement('section');
    host.className = 'new-modt-task-config';
    host.setAttribute('data-region', 'modtnew.task_config');
    host.innerHTML = '<div class="new-modt-task-config__label">Cấu hình để Task chạy'
      + '<span class="new-modt-task-config__draft">· mở phần nào, khai phần đó</span></div>'
      + MODULES.map(function (module) { return renderModule(module, taskKey); }).join('')
      + renderConditionPicker();

    host.querySelectorAll('.new-modt-config-module').forEach(function (details) {
      details.addEventListener('toggle', function () {
        openState[taskKey][details.getAttribute('data-module-id')] = details.open;
      });
    });

    host.querySelectorAll('[data-action="module-help"]').forEach(function (button) {
      button.addEventListener('click', function (event) {
        event.preventDefault();
        event.stopPropagation();
        var moduleElement = button.closest('.new-modt-config-module');
        var panel = moduleElement.querySelector('[data-role="module-help-panel"]');
        var shouldOpen = !panel.classList.contains('is-open');
        moduleElement.open = true;
        panel.classList.toggle('is-open', shouldOpen);
        button.setAttribute('aria-expanded', shouldOpen ? 'true' : 'false');
      });
    });

    var picker = host.querySelector('.new-modt-condition-picker');
    var pickerValues = picker.querySelector('.new-modt-condition-picker__values');
    var pickerSearch = picker.querySelector('.new-modt-condition-picker__search');
    var pickerChoose = picker.querySelector('[data-action="condition-choose"]');
    var pickerNote = picker.querySelector('[data-role="condition-picker-note"]');
    var activeConditionTrigger = null;
    var selectedField = null;
    var selectedValue = null;

    function closeConditionPicker() {
      picker.classList.remove('is-open');
      picker.setAttribute('aria-hidden', 'true');
    }

    function bindValueRows() {
      pickerValues.querySelectorAll('[data-value-code]').forEach(function (row) {
        row.addEventListener('click', function () {
          pickerValues.querySelectorAll('.is-selected').forEach(function (item) {
            item.classList.remove('is-selected');
          });
          row.classList.add('is-selected');
          selectedValue = {
            code: row.getAttribute('data-value-code'),
            name: row.getAttribute('data-value-name')
          };
          pickerChoose.disabled = false;
        });
      });
    }

    function showValues(query) {
      pickerValues.innerHTML = renderConditionValueRows(selectedField ? selectedField.code : '', query);
      selectedValue = null;
      pickerChoose.disabled = true;
      bindValueRows();
    }

    function bindAssignmentRuleRow(row) {
      row.querySelectorAll('.new-modt-assignment-rule-table__connector-toggle').forEach(function (button) {
        button.addEventListener('click', function () {
          var nextValue = button.getAttribute('data-value') === 'AND' ? 'OR' : 'AND';
          button.setAttribute('data-value', nextValue);
          button.setAttribute('aria-label', 'Đang chọn ' + nextValue + '; bấm để đổi AND hoặc OR');
          button.textContent = nextValue.toLowerCase();
        });
      });
      row.querySelectorAll('.new-modt-assignment-rule-table__condition-trigger').forEach(function (button) {
        button.addEventListener('click', function () {
        activeConditionTrigger = button;
        selectedField = null;
        selectedValue = null;
        pickerSearch.value = '';
        pickerSearch.disabled = true;
        pickerChoose.disabled = true;
        pickerNote.classList.remove('is-error');
        pickerNote.textContent = 'ALL = mọi giá trị hợp lệ · nguồn dài tìm và phân trang trên máy chủ.';
        picker.querySelectorAll('.is-selected').forEach(function (item) {
          item.classList.remove('is-selected');
        });
        pickerValues.innerHTML = '<tr><td colspan="3" class="new-modt-condition-picker__empty">Chọn một trường ở bảng bên trái</td></tr>';
        picker.classList.add('is-open');
        picker.setAttribute('aria-hidden', 'false');
        });
      });
    }

    host.querySelectorAll('.new-modt-assignment-rule-table__body .t1-mapping-table__row').forEach(bindAssignmentRuleRow);

    picker.querySelectorAll('[data-field-code]').forEach(function (row) {
      row.addEventListener('click', function () {
        picker.querySelectorAll('[data-field-code].is-selected').forEach(function (item) {
          item.classList.remove('is-selected');
        });
        row.classList.add('is-selected');
        selectedField = CONDITION_FIELD_CATALOG.find(function (field) {
          return field.code === row.getAttribute('data-field-code');
        });
        pickerSearch.disabled = false;
        pickerSearch.value = '';
        showValues('');
        pickerSearch.focus();
      });
    });

    pickerSearch.addEventListener('input', function () {
      showValues(pickerSearch.value);
    });

    picker.querySelectorAll('[data-action="condition-close"]').forEach(function (button) {
      button.addEventListener('click', closeConditionPicker);
    });

    pickerChoose.addEventListener('click', function () {
      if (!activeConditionTrigger || !selectedField || !selectedValue) return;
      var conditionKeys = ['anchor_1', 'anchor_2', 'condition'];
      var currentIndex = conditionKeys.indexOf(activeConditionTrigger.getAttribute('data-condition-key'));
      var activeRuleRow = activeConditionTrigger.closest('.t1-mapping-table__row');
      var conditionTriggers = conditionKeys.map(function (key) {
        return activeRuleRow.querySelector('.new-modt-assignment-rule-table__condition-trigger[data-condition-key="' + key + '"]');
      });
      var previousOrder = currentIndex > 0
        ? Number(conditionTriggers[currentIndex - 1].getAttribute('data-field-order') || 0)
        : 0;
      var nextOrder = currentIndex < conditionTriggers.length - 1
        ? Number(conditionTriggers[currentIndex + 1].getAttribute('data-field-order') || 0)
        : 0;
      var isExactDuplicate = conditionTriggers.some(function (trigger, index) {
        return index !== currentIndex
          && trigger.getAttribute('data-field-code') === selectedField.code
          && trigger.getAttribute('data-value-code') === selectedValue.code;
      });
      var hasAllOverlap = conditionTriggers.some(function (trigger, index) {
        var existingValue = trigger.getAttribute('data-value-code');
        return index !== currentIndex
          && trigger.getAttribute('data-field-code') === selectedField.code
          && (existingValue === 'ALL' || selectedValue.code === 'ALL');
      });
      if (isExactDuplicate) {
        pickerNote.classList.add('is-error');
        pickerNote.textContent = 'Điều kiện này đã được chọn trong cùng nguyên tắc.';
        return;
      }
      if (hasAllOverlap) {
        pickerNote.classList.add('is-error');
        pickerNote.textContent = 'ALL đã bao phủ toàn bộ trường này; không chọn thêm giá trị cùng trường trong một nguyên tắc.';
        return;
      }
      if ((previousOrder && selectedField.order < previousOrder)
        || (nextOrder && selectedField.order > nextOrder)) {
        pickerNote.classList.add('is-error');
        pickerNote.textContent = 'Sai thứ tự bảng gốc. Hãy chọn lại Điều kiện 1 → 2 → 3 theo thứ tự tăng dần.';
        return;
      }
      activeConditionTrigger.querySelector('.new-modt-assignment-rule-table__condition-field').textContent =
        selectedField.name + ' =';
      activeConditionTrigger.querySelector('.new-modt-assignment-rule-table__condition-value').textContent =
        selectedValue.code + ' · ' + selectedValue.name;
      activeConditionTrigger.setAttribute('data-field-order', String(selectedField.order));
      activeConditionTrigger.setAttribute('data-field-code', selectedField.code);
      activeConditionTrigger.setAttribute('data-value-code', selectedValue.code);
      activeConditionTrigger.setAttribute('data-match-mode', selectedValue.code === 'ALL' ? 'ALL' : 'EXACT');
      closeConditionPicker();
    });

    host.querySelectorAll('[data-action="add-rule"]').forEach(function (button) {
      button.addEventListener('click', function () {
        var moduleElement = button.closest('.new-modt-config-module');
        var note = button.parentElement.querySelector('.new-modt-config-module__foot-note');
        if (moduleElement && moduleElement.getAttribute('data-module-id') === 'assignment_rule') {
          var tableBody = moduleElement.querySelector('.new-modt-assignment-rule-table__body');
          var currentCount = tableBody.querySelectorAll('.t1-mapping-table__row').length;
          if (currentCount >= 50) {
            note.textContent = '50/50';
            return;
          }
          var nextCode = 'NTGV' + String(currentCount + 1).padStart(2, '0');
          tableBody.insertAdjacentHTML('beforeend', renderAssignmentRuleRow(nextCode));
          bindAssignmentRuleRow(tableBody.lastElementChild);
          note.textContent = String(currentCount + 1) + '/50';
          return;
        }
        note.textContent = 'Bản nháp: nút thêm dòng sẽ được nối sau khi chốt cách khai báo.';
      });
    });

    root.appendChild(host);
  }

  window.NEW_MODT_TASK_CONFIG = {
    mount: mount,
    schema: MODULES,
    conditionFieldCatalog: CONDITION_FIELD_CATALOG,
    conditionAllCode: 'ALL',
    conditionAllSemantics: 'Mọi giá trị hợp lệ của trường; không bao gồm giá trị trống. Điều kiện cụ thể ưu tiên hơn ALL.'
  };
}());
