import subprocess,json,time
cases=[('B01','field.so_phut','field.so_phut','exact key'),('B02','phút','field.so_phut','exact alias'),('B03','PHÚT','field.so_phut','case'),('B04','  phút  ','field.so_phut','whitespace'),('B05','so phut thuc hien','field.so_phut','no accent'),('B06','WF','workflow','abbreviation'),('B07','Số phút thưc hiện','field.so_phut','typo'),('B08','thời lượng một task','field.so_phut','same meaning new wording'),('B09','Tổng phút','field.tong_phut_congdon','near family distinct'),('B10','ngày tạo','field.ngay_lap','alias date'),('B11','ngày sửa đổi',None,'different meaning'),('B12','trạng thái','field.trang_thai_quy_trinh','ambiguous runtime-state brief'),('B13','phút',None,'requested unit hours'),('B14','người nhận hàng','role.nguoi_thuc_hien','meaning in description'),('B15','khoi','org.khoi','approved unaccent alias'),('B16','CCCD',None,'catalog coverage unknown')]
def lit(s):return "'"+s.replace("'","''")+"'"
values=','.join('('+lit(i)+','+lit(q)+')' for i,q,_,_ in cases)
q="""WITH cases(case_id,q) AS (VALUES %s), found AS (
SELECT b.case_id,c.id,c.concept_key,CASE WHEN c.concept_key=b.q THEN 0 WHEN lower(btrim(c.ten))=lower(btrim(b.q)) THEN 1 ELSE 2 END AS rank
FROM cases b JOIN sua_re_khai_niem_nhap c ON c.status='active' AND (c.concept_key=b.q OR lower(btrim(c.ten))=lower(btrim(b.q)) OR EXISTS (SELECT 1 FROM jsonb_array_elements_text(c.aliases) a WHERE lower(btrim(a))=lower(btrim(b.q))))
) SELECT json_agg(row_to_json(r)) FROM (SELECT b.case_id,b.q,coalesce((SELECT json_agg(z.concept_key) FROM (SELECT f.concept_key FROM found f WHERE f.case_id=b.case_id ORDER BY f.rank,f.id LIMIT 5)z),'[]') AS candidates FROM cases b ORDER BY case_id)r"""%values
start=time.perf_counter()
r=subprocess.run(['docker','exec','postgres','psql','-X','-U','directus','-d','directus','-At','-v','ON_ERROR_STOP=1','-c','BEGIN READ ONLY; SET LOCAL statement_timeout=8000; '+q+'; ROLLBACK;'],capture_output=True,text=True)
elapsed=(time.perf_counter()-start)*1000
print(json.dumps({'fixture':'native SQL exact baseline on provisional concept metadata; NOT T1 deployment','cases':cases,'elapsed_docker_psql_ms':round(elapsed,2),'stdout':r.stdout,'stderr':r.stderr},ensure_ascii=False))
