Mẫu Host đã chạy thật (PostgreSQL 16.15 và 18.4, đủ 2.148 dòng balo, 03/10)

Chạy sau khi nhãn đã hợp lệ; chạy sớm thì PG từ chối và lùi sạch. Cột khoá tính từ khoa_nhan_dien, phải ghi rõ STORED (PG 18 mặc định là cột ảo, không gắn khoá ngoại được).

ALTER TABLE public.balo_thuc_the
  ADD COLUMN k_khoi text GENERATED ALWAYS AS ('KHOI'::text) STORED,
  ADD COLUMN k_vai text GENERATED ALWAYS AS (CASE split_part(khoa_nhan_dien, ':', 1) WHEN 'table' THEN 'VAI_BANG' WHEN 'function' THEN 'VAI_HAM' WHEN 'view' THEN 'VAI_X' WHEN 'trigger' THEN 'VAI_X' END) STORED,
  ADD COLUMN k_song text GENERATED ALWAYS AS (CASE split_part(khoa_nhan_dien, ':', 1) WHEN 'table' THEN 'SONG_BANG' WHEN 'function' THEN 'SONG_KHONGDO' WHEN 'view' THEN 'SONG_KHONGDO' WHEN 'trigger' THEN 'SONG_TRIGGER' END) STORED,
  ADD COLUMN k_ketluan text GENERATED ALWAYS AS ('KETLUAN'::text) STORED;
ALTER TABLE public.balo_thuc_the
  ADD CONSTRAINT balo_nhan_khoi_fk    FOREIGN KEY (k_khoi, nhom)        REFERENCES public.balo_tu_dien_nhan (nhan, gia_tri),
  ADD CONSTRAINT balo_nhan_vai_fk     FOREIGN KEY (k_vai, loai)         REFERENCES public.balo_tu_dien_nhan (nhan, gia_tri),
  ADD CONSTRAINT balo_nhan_song_fk    FOREIGN KEY (k_song, active)      REFERENCES public.balo_tu_dien_nhan (nhan, gia_tri),
  ADD CONSTRAINT balo_nhan_ketluan_fk FOREIGN KEY (k_ketluan, ket_luan) REFERENCES public.balo_tu_dien_nhan (nhan, gia_tri);

Hai truy vấn trang đã qua bộ lọc của dot-report-publish và khớp từng giá trị theo cách so của dot-table-ui-verify. Cột đầu phải là stt duy nhất, các cột khác tên từ 4 ký tự, không trả cột kiểu ngày.

-- Luật nhãn
SELECT row_number() OVER (ORDER BY thu_tu, id)::int AS stt,
       nhan, gia_tri, ten_hien_thi, ap_cho, cau_thu, vi_du
FROM balo_tu_dien_nhan
ORDER BY thu_tu, id

-- Balo theo khối
SELECT row_number() OVER (ORDER BY COALESCE(b.nhom, chr(1)))::int AS stt,
       COALESCE(b.nhom, '(chưa dán)') AS khoi,
       count(*) AS so_vat,
       count(*) FILTER (WHERE b.loai_census = 'table') AS so_bang,
       count(*) FILTER (WHERE b.loai_census = 'view') AS so_view,
       count(*) FILTER (WHERE b.loai_census = 'function') AS so_ham,
       count(*) FILTER (WHERE b.loai_census = 'trigger') AS so_trigger,
       count(*) FILTER (WHERE b.loai_census = 'table' AND b.active = 'Rỗng') AS bang_rong,
       count(*) FILTER (WHERE b.loai_census = 'table' AND b.active = 'Đứng yên') AS bang_dung_yen,
       count(*) FILTER (WHERE b.loai_census = 'table' AND b.active = 'Tự quay') AS bang_tu_quay,
       count(*) FILTER (WHERE b.loai_census = 'table' AND b.active = 'Có ghi') AS bang_co_ghi,
       count(*) FILTER (WHERE b.nhom = 'CHUA-RO' OR b.loai = 'CHUA-RO' OR b.active = 'CHUA-RO') AS chua_ro,
       count(*) FILTER (WHERE b.active = 'Không đo được') AS khong_do_duoc,
       sum(b.dinh) AS tong_dinh,
       CASE WHEN count(b.ket_luan) = 0 THEN NULL
            WHEN count(DISTINCT b.ket_luan) = 1 AND count(b.ket_luan) = count(*) THEN max(b.ket_luan)
            ELSE 'lẫn' END AS ket_luan
FROM balo_thuc_the b
WHERE NOT b.da_mat
GROUP BY b.nhom
ORDER BY 1
